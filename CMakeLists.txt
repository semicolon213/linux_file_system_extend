project(filescope VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

find_package(PkgConfig REQUIRED)
pkg_check_modules(GTKMM REQUIRED IMPORTED_TARGET gtkmm-3.0)
pkg_check_modules(GLIBMM REQUIRED IMPORTED_TARGET glibmm-2.4)
pkg_check_modules(GIOMM REQUIRED IMPORTED_TARGET giomm-2.4)
pkg_check_modules(SIGC REQUIRED IMPORTED_TARGET sigc++-2.0)
pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0)
pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET glib-2.0)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0)
pkg_check_modules(SQLITE3 REQUIRED IMPORTED_TARGET sqlite3)
pkg_check_modules(AYATANA REQUIRED IMPORTED_TARGET ayatana-appindicator3-0.1)
set_property(TARGET PkgConfig::AYATANA PROPERTY SYSTEM TRUE)

function(filescope_configure_target target)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(FILESCOPE_SANITIZER)
    target_compile_options(${target} PRIVATE -fsanitize=${FILESCOPE_SANITIZER} -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=${FILESCOPE_SANITIZER} -fno-omit-frame-pointer)
    if(FILESCOPE_SANITIZER STREQUAL "undefined")
      target_compile_options(${target} PRIVATE -fno-sanitize-recover=undefined)
      target_link_options(${target} PRIVATE -fno-sanitize-recover=undefined)
    endif()
  endif()
endfunction()

function(filescope_add_test name target)
  if(FILESCOPE_SANITIZER STREQUAL "thread")
    find_program(SETARCH_EXECUTABLE setarch REQUIRED)
    add_test(NAME ${name}
             COMMAND ${SETARCH_EXECUTABLE} ${CMAKE_SYSTEM_PROCESSOR} -R $<TARGET_FILE:${target}>)
  else()
    add_test(NAME ${name} COMMAND ${target})
  endif()
endfunction()

add_library(filescope_info STATIC src/app_info.cpp)
target_include_directories(filescope_info PUBLIC ${PROJECT_SOURCE_DIR}/src)
target_compile_definitions(filescope_info PRIVATE FILESCOPE_VERSION="${PROJECT_VERSION}")

foreach(module core job fs mon nav dash diff his search trash ui)
  if(EXISTS "${PROJECT_SOURCE_DIR}/src/${module}/CMakeLists.txt")
    add_subdirectory(src/${module})
  endif()
endforeach()

add_executable(filescope src/main.cpp)
target_link_libraries(filescope PRIVATE filescope_info PkgConfig::GTKMM)
if(TARGET filescope_ui)
  target_link_libraries(filescope PRIVATE filescope_ui)
endif()

filescope_configure_target(filescope_info)
filescope_configure_target(filescope)
