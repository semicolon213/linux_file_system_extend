file(GLOB integration_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/test_*.c")

foreach(source IN LISTS integration_sources)
  get_filename_component(target ${source} NAME_WE)
  if(NOT target MATCHES "^test_([^_]+)_")
    message(FATAL_ERROR "Integration test name must be test_<module>_<topic>.c: ${source}")
  endif()
  set(module ${CMAKE_MATCH_1})
  if(NOT TARGET filescope_${module})
    message(FATAL_ERROR "Integration test ${target} has no filescope_${module} target")
  endif()

  add_executable(${target} ${source})
  target_link_libraries(${target} PRIVATE filescope_${module} filescope_test_support)
  filescope_configure_target(${target})
  filescope_add_test(${target} ${target})
  if(target MATCHES "_stress_")
    set(label stress)
  elseif(target MATCHES "_bench_")
    set(label benchmark)
  else()
    set(label integration)
  endif()
  set_tests_properties(${target} PROPERTIES LABELS ${label})
endforeach()
